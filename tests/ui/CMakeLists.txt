cmake_minimum_required(VERSION 3.22)
set(PLATFORM_DESKTOP ON CACHE BOOL "" FORCE)
set(USE_GLFW OFF CACHE BOOL "" FORCE)
set(USE_SDL2 ON CACHE BOOL "" FORCE)
set(USE_SYSTEM_SDL2 ON CACHE BOOL "" FORCE)
set(USE_GL3 ON CACHE BOOL "" FORCE)
set(USE_LIBROMFS OFF CACHE BOOL "" FORCE)
get_filename_component(SWITCHMUSIC_ROOT ../.. ABSOLUTE)
set(BRLS_RESOURCES_DIR ${CMAKE_BINARY_DIR})
include(${SWITCHMUSIC_ROOT}/third_party/borealis/library/cmake/commonOption.cmake)
include(${SWITCHMUSIC_ROOT}/third_party/borealis/library/cmake/toolchain.cmake)
project(SwitchMusicUIPreview LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
add_subdirectory(${SWITCHMUSIC_ROOT}/third_party/borealis/library borealis)
find_package(PkgConfig REQUIRED)
pkg_check_modules(CURL REQUIRED IMPORTED_TARGET libcurl)
add_executable(switchmusic-ui-preview preview.cpp ${SWITCHMUSIC_ROOT}/common/src/client.cpp ${SWITCHMUSIC_ROOT}/common/src/ipc.cpp)
target_include_directories(switchmusic-ui-preview PRIVATE ${SWITCHMUSIC_ROOT}/common/include ${SWITCHMUSIC_ROOT}/third_party/json)
target_link_libraries(switchmusic-ui-preview PRIVATE borealis PkgConfig::CURL)
file(COPY ${SWITCHMUSIC_ROOT}/third_party/borealis/resources/ DESTINATION ${CMAKE_BINARY_DIR}/resources)
add_custom_target(ui-resources ALL
    COMMAND ${CMAKE_COMMAND} -E copy_directory ${SWITCHMUSIC_ROOT}/nro/resources ${CMAKE_BINARY_DIR}/resources)
add_dependencies(switchmusic-ui-preview ui-resources)
